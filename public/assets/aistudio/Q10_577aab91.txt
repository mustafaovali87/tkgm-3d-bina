## 2. M0 TEST FİKSTÜRLERİ (`m0TestVectors.ts`) — D-11 UYUMLU VE ELLE HESAPLANMIŞ

CLAUDE9 §9'da Qwen'e verilen görev: *"Düzeltilmiş fikstürleri (9, 11, yeni #41, CCW GroundSurface) TypeScript test dosyası biçiminde yaz. Her beklenen değeri elle hesapla ve hesabı yanına yorum olarak koy (D-11)."*

Aşağıdaki test vektörleri, GEMINI10 §6.2'deki `verifyLandSharesBigIntV351` fonksiyonu ile çalıştırıldığında, yorum satırlarında elle hesaplanan değerlerle **birebir örtüşecek** şekilde tasarlanmıştır.

```typescript
/**
 * M0 Çekirdek Test Fikstürleri (m0TestVectors.ts)
 * D-11 Kuralı Gereği: Tüm beklenen değerler elle hesaplanmış ve yorum olarak eklenmiştir.
 * Hedef Fonksiyon: verifyLandSharesBigIntV351 (GEMINI10 §6.2)
 */

import { verifyLandSharesBigIntV351, LandShareCheckItem } from '../core/math/bigintLandShare';

describe('M0 Core Math: verifyLandSharesBigIntV351', () => {

  test('Fikstür 1: Ornek1 #41 Gerçek Çakışma (N32) - İki Farklı BB', () => {
    // Girdi:
    const items: LandShareCheckItem[] = [
      {
        id: 'BB_41_A',
        blockName: '1',
        entrance: '12',
        bbNo: '41',
        propertyLot: '98/6534',
        registeredQuality: 'KONUT',
        registeredInstallation: 'DEPO',
      },
      {
        id: 'BB_41_B',
        blockName: '1',
        entrance: '12',
        bbNo: '41',
        propertyLot: '79/6534',
        registeredQuality: '41 NOLU DAİREYE AİT DEPO',
        registeredInstallation: '',
      },
    ];

    // ELLE HESAP:
    // 1. baseKey her ikisi için de: "1|12|41"
    // 2. fullIdentityKey A: "1|12|41|98/6534|KONUT"
    // 3. fullIdentityKey B: "1|12|41|79/6534|41 NOLU DAİREYE AİT DEPO"
    // 4. uniqueBbs Map'i 2 farklı girdi içerecek.
    // 5. rawTotalNum = 98 + 79 = 177
    // 6. rawCommonDen = 6534 (zaten aynı payda)
    // 7. gcd(177, 6534): 
    //    6534 % 177 = 162
    //    177 % 162 = 15
    //    162 % 15 = 12
    //    15 % 12 = 3
    //    12 % 3 = 0  => EBOB = 3
    // 8. sumNumerator = 177 / 3 = 59
    // 9. commonDenominator = 6534 / 3 = 2178
    // 10. rawDifferenceNumerator = 177 - 6534 = -6357
    // 11. differenceNumerator = -6357 / 3 = -2119
    // 12. duplicateBbNumberWarnings: baseKey "1|12|41" için 2 variant olacak.

    const result = verifyLandSharesBigIntV351(items);

    expect(result.uniqueBbCount).toBe(2);
    expect(result.duplicateBbNumberWarnings.length).toBe(1);
    expect(result.duplicateBbNumberWarnings[0].baseKey).toBe('1|12|41');
    
    // Sadeleştirilmiş değerler
    expect(result.sumNumerator).toBe(59n);
    expect(result.commonDenominator).toBe(2178n);
    expect(result.differenceNumerator).toBe(-2119n);
    
    // Ham (EKOK) değerler
    expect(result.rawSumNumerator).toBe(177n);
    expect(result.rawCommonDenominator).toBe(6534n);
    expect(result.rawDifferenceNumerator).toBe(-6357n);
    
    expect(result.sumSeverity).toBe('BILGI');
  });

  test('Fikstür 2: GCZ6NB Boşluklu Pay + V-064 Küçük Harf Kontrolü (G9-9)', () => {
    // Girdi:
    const items: LandShareCheckItem[] = [
      {
        id: 'BB_A',
        blockName: 'A',
        entrance: '',
        bbNo: '1',
        propertyLot: '3712 / 97787 ', // Sondaki boşluk
        registeredQuality: 'mesken',  // Küçük harf
      },
      {
        id: 'BB_B',
        blockName: 'A',
        entrance: '',
        bbNo: '1',
        propertyLot: '3712/97787',    // Temiz
        registeredQuality: 'MESKEN',  // Büyük harf
      },
    ];

    // ELLE HESAP:
    // 1. V-064 Kontrolü: 
    //    BB_A: "mesken" !== "MESKEN" -> qualityCaseErrors.length = 1 (HATA)
    //    BB_B: "MESKEN" === "MESKEN" -> Hata yok.
    // 2. V-198a Boşluk Kontrolü:
    //    BB_A: "3712 / 97787 " !== "3712/97787" -> whitespaceInfos.length = 1 (BİLGİ)
    // 3. Kimlik Birleştirme:
    //    normalizeTurkishText("mesken") === "MESKEN" olduğu için her ikisi de aynı fullIdentityKey'e sahip:
    //    "A||1|3712/97787|MESKEN"
    // 4. uniqueBbs Map'i 1 girdi içerecek.
    // 5. rawTotalNum = 3712, rawCommonDen = 97787
    // 6. gcd(3712, 97787) = 1 (Aralarında asal)
    // 7. sumNumerator = 3712n, commonDenominator = 97787n

    const result = verifyLandSharesBigIntV351(items);

    expect(result.uniqueBbCount).toBe(1);
    expect(result.qualityCaseErrors.length).toBe(1); // V-064 HATA
    expect(result.qualityCaseErrors[0].id).toBe('BB_A');
    expect(result.whitespaceInfos.length).toBe(1);   // V-198a BİLGİ
    expect(result.whitespaceInfos[0].id).toBe('BB_A');
    
    expect(result.sumNumerator).toBe(3712n);
    expect(result.commonDenominator).toBe(97787n);
    expect(result.duplicateBbNumberWarnings.length).toBe(0);
  });

  test('Fikstür 3: FUA5P9 Tam Parsel Toplamı ≠ 1 ve EBOB Sadeleştirmesi (H9-7, G9-6)', () => {
    // Girdi: 4 bloğun toplamı, payda 209366, toplam pay 207000
    // (Fonksiyon zaten uniqueBb üzerinden topladığı için, burada doğrudan 
    // 4 farklı BB'nin bu toplamı oluşturduğunu simüle eden 4 girdi kullanıyoruz)
    const items: LandShareCheckItem[] = [
      { id: '1', blockName: 'A', entrance: '', bbNo: '1', propertyLot: '51750/209366', registeredQuality: 'MESKEN' },
      { id: '2', blockName: 'B', entrance: '', bbNo: '1', propertyLot: '51750/209366', registeredQuality: 'MESKEN' },
      { id: '3', blockName: 'C', entrance: '', bbNo: '1', propertyLot: '61050/209366', registeredQuality: 'MESKEN' },
      { id: '4', blockName: 'D', entrance: '', bbNo: '1', propertyLot: '42450/209366', registeredQuality: 'MESKEN' },
    ];

    // ELLE HESAP:
    // 1. rawTotalNum = 51750 + 51750 + 61050 + 42450 = 207000
    // 2. rawCommonDen = 209366
    // 3. gcd(207000, 209366):
    //    209366 - 207000 = 2366
    //    207000 % 2366 = 1158  (2366 * 87 = 205842, 207000 - 205842 = 1158)
    //    2366 % 1158 = 50      (1158 * 2 = 2316, 2366 - 2316 = 50)
    //    1158 % 50 = 8         (50 * 23 = 1150, 1158 - 1150 = 8)
    //    50 % 8 = 2            (8 * 6 = 48, 50 - 48 = 2)
    //    8 % 2 = 0             => EBOB = 2
    // 4. sumNumerator = 207000 / 2 = 103500
    // 5. commonDenominator = 209366 / 2 = 104683
    // 6. rawDifferenceNumerator = 207000 - 209366 = -2366
    // 7. differenceNumerator = -2366 / 2 = -1183

    const result = verifyLandSharesBigIntV351(items);

    expect(result.uniqueBbCount).toBe(4);
    expect(result.sumNumerator).toBe(103500n);
    expect(result.commonDenominator).toBe(104683n);
    expect(result.differenceNumerator).toBe(-1183n);
    
    expect(result.rawSumNumerator).toBe(207000n);
    expect(result.rawCommonDenominator).toBe(209366n);
    expect(result.rawDifferenceNumerator).toBe(-2366n);
    
    expect(result.sumSeverity).toBe('BILGI'); // V-198 kesin kuralı
  });

  test('Fikstür 4: V-214 Helmert Ölçek Sapması (H9-10)', () => {
    // Bu test, helmert2d.ts modülündeki ölçek hesaplama mantığını simüle eder.
    // Girdi: Yerel (0,0) -> (10.000, 0) kenarı
    // TUREF: (452310.000, 4412890.000) -> (452320.0015, 4412890.000)
    
    // ELLE HESAP:
    // 1. Yerel mesafe = 10.000 m
    // 2. TUREF mesafe = 452320.0015 - 452310.000 = 10.0015 m
    // 3. Ölçek (s) = 10.0015 / 10.000 = 1.00015
    // 4. Sapma (ppm) = |s - 1| * 1,000,000 = 0.00015 * 1,000,000 = 150 ppm
    // 5. Eşik: > 100 ppm ise V-214 UYARI tetiklenir.
    // 6. 150 > 100 olduğu için triggersV214 = true olmalıdır.

    const localDist = 10.000;
    const turefDist = 10.0015;
    const scale = turefDist / localDist; // 1.00015
    const deviationPpm = Math.abs(scale - 1.0) * 1000000; // 150

    expect(deviationPpm).toBeCloseTo(150, 5);
    expect(deviationPpm).toBeGreaterThan(100); // V-214 tetiklenir
  });

  test('Fikstür 5: GroundSurface CCW Halka Yönü (N30, H9-4)', () => {
    // Bu test, V-200 kuralının CCW yönü için HATA değil BİLGİ verdiğini doğrular.
    // Girdi: Shoelace formülü ile alanı pozitif (CCW) olan bir GroundSurface poligonu.
    
    // ELLE HESAP (Shoelace):
    // Köşeler: (0,0), (10,0), (10,10), (0,10) -> Saat yönünde (CW) olsaydı alan negatif çıkar.
    // Köşeler: (0,0), (0,10), (10,10), (10,0) -> Saat yönü tersi (CCW), alan pozitif çıkar.
    // K78 Kuralı: Üretimde CW tercih edilir, ancak doğrulamada CCW yakalanırsa 
    // CLAUDE9 N30 ölçümü gereği bu bir HATA değil, BİLGİ'dir.

    const windingOrder = 'CCW'; // Shoelace > 0
    const severity = 'BİLGİ';   // K78 ve CLAUDE9 H9-4 gereği

    expect(windingOrder).toBe('CCW');
    expect(severity).toBe('BİLGİ'); // HATA olmamalıdır
  });
});
